# 间距与布局

TOPEMS 设计系统提供完整的间距、圆角、阴影、断点和层级体系，所有值源自 `@topems/tokens`。此外，`semanticTokens` 提供语义化映射，描述"在做什么"而非"是多少 px"。

import React from 'react';

{/* 间距可视条组件 */}
const SpacingBar = ({ name, value, color = 'primary' }) => (
  <Box sx={{ display: 'flex', alignItems: 'center', gap: 2, mb: 1.5 }}>
    <Chip label={name} size="small" sx={{ minWidth: 56 }} color={color === 'primary' ? 'primary' : 'secondary'} />
    <Box sx={{ flex: 1 }}>
      <Box sx={{
        height: 32,
        width: `${value}px`,
        maxWidth: '100%',
        backgroundColor: color === 'primary' ? designTokens.colors.primary.main : designTokens.colors.secondary.main,
        borderRadius: designTokens.borderRadius.sm,
        display: 'flex',
        alignItems: 'center',
        justifyContent: 'center',
        color: '#ffffff',
        fontSize: '0.75rem',
        fontFamily: 'monospace',
        fontWeight: 600,
        minWidth: 28,
      }}>
        {value}px
      </Box>
    </Box>
    <Typography variant="caption" sx={{ fontFamily: 'monospace', color: 'text.secondary', flexShrink: 0 }}>
      {value === 0 ? '0' : `${value / 8}u`}
    </Typography>
  </Box>
);

{/* 圆角可视组件 */}
const RadiusBox = ({ name, value }) => (
  <Box sx={{ display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 1 }}>
    <Box sx={{
      width: 64,
      height: 64,
      borderRadius: value === 9999 ? '50%' : `${value}px`,
      border: `2px solid ${designTokens.colors.primary.main}`,
      backgroundColor: designTokens.colors.primary[50],
    }} />
    <Typography variant="caption" fontWeight="bold">{name}</Typography>
    <Typography variant="caption" sx={{ fontFamily: 'monospace', color: 'text.secondary' }}>
      {value === 9999 ? '9999 (full)' : `${value}px`}
    </Typography>
  </Box>
);

{/* 阴影可视组件 */}
const ShadowBox = ({ name, value }) => (
  <Box sx={{ display: 'flex', flexDirection: 'column', alignItems: 'center', gap: 1 }}>
    <Box sx={{
      width: 80,
      height: 56,
      borderRadius: designTokens.borderRadius.lg,
      boxShadow: value,
      backgroundColor: '#ffffff',
      border: `1px solid ${designTokens.colors.border.light}`,
    }} />
    <Typography variant="caption" fontWeight="bold">{name}</Typography>
  </Box>
);
