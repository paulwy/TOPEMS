# TOPEMS 页面实施指南 v1.0.0

## 概述

本文档定义了TOPEMS系统中所有页面创建的标准规范和最佳实践。通过遵循这些规范,可以确保:
- 所有页面具有统一的用户体验
- 代码结构清晰、可维护性强
- 符合React + TypeScript + Material-UI技术栈
- 支持响应式设计和主题切换

---

## 页面类型概览

TOPEMS系统采用四种核心页面模版:

| 页面类型 | 模版组件 | 适用场景 | 路由模式 |
|---------|---------|---------|---------|
| **仪表盘** | `DashboardTemplate` | 数据概览、统计图表 | `/module` |
| **列表页** | `ListPageTemplate` | 数据列表、搜索筛选 | `/module/entities` |
| **详情页** | `DetailPageTemplate` | 单条数据详情、Tab切换 | `/module/entities/:id` |
| **表单页** | `FormPageTemplate` | 新建/编辑数据 | `/module/entities/create` 或 `/module/entities/:id/edit` |

---

## 一、仪表盘页面 (DashboardTemplate)

### 1.1 适用场景

- 模块首页概览
- 数据统计分析
- 快捷操作入口
- 最近事项展示

### 1.2 路由规范

```
路由: /module (例: /finance, /hrm, /crm)
```

## 二、详情页面 (DetailPageTemplate)

### 2.1 适用场景

- 查看单条数据的完整信息
- 包含多个信息维度的展示
- 需要Tab切换不同内容区域
- 需要编辑、删除等操作

### 2.2 路由规范

```
路由: /module/entities/:id (例: /finance/vouchers/:id, /crm/customers/:id)
```

### 2.3 完整示例

**文件路径**: `/apps/web/src/pages/finance/VoucherDetailPage.tsx`

## 三、表单页面 (FormPageTemplate)

### 3.1 适用场景

- 新建数据记录
- 编辑已有数据
- 复杂的数据录入表单

### 3.2 路由规范

```
新建: /module/entities/create (例: /finance/vouchers/create)
编辑: /module/entities/:id/edit (例: /finance/vouchers/:id/edit)
```

### 3.3 完整示例

**文件路径**: `/apps/web/src/pages/finance/VoucherCreatePage.tsx`

```tsx
import {
  Card,
  CardContent,
  TextField,
  Typography,
  Box,
  Stack,
  Divider,
  IconButton,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
  Autocomplete,
  Alert,
  Button,
} from "@mui/material";
import React, { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import {
  ArrowLeft,
  Plus,
  Trash2,
  Save,
  X,
  Calculator,
  AlertCircle,
} from "lucide-react";
import { FormPageTemplate } from "@/components/templates";
import { voucherApi, accountingSubjectApi, type AccountingSubject, type VoucherItemRequest } from "@/api/erp";
import { useAppStore } from "@/store";

const VoucherCreatePage: React.FC = () => {
  const navigate = useNavigate();
  const addNotification = useAppStore((state) => state.addNotification);

  const [subjects, setSubjects] = useState<AccountingSubject[]>([]);
  const [loading, setLoading] = useState(false);
  const [submitting, setSubmitting] = useState(false);

  const [formData, setFormData] = useState({
    voucher_number: "",
    voucher_date: "",
    period: "",
    attachment_count: 0,
    summary: "",
  });

  const [voucherItems, setVoucherItems] = useState<VoucherItemRequest[]>([]);

  useEffect(() => {
    loadSubjects();
    initializeForm();
  }, []);

  const loadSubjects = async () => {
    try {
      setLoading(true);
      const result = await accountingSubjectApi.list();
      if (result.success && result.data) {
        setSubjects(result.data.items || []);
      }
    } catch (err) {
      console.error("加载会计科目失败:", err);
    } finally {
      setLoading(false);
    }
  };

  const initializeForm = () => {
    const today = new Date();
    const period = `${today.getFullYear()}-${String(today.getMonth() + 1).padStart(2, "0")}`;
    setFormData((prev) => ({ ...prev, period }));
    addVoucherItem();
    addVoucherItem();
  };

  const handleFieldChange = (field: string, value: any) => {
    setFormData((prev) => ({ ...prev, [field]: value }));
  };

  const addVoucherItem = () => {
    const newItem: VoucherItemRequest = {
      line_number: voucherItems.length + 1,
      subject_code: "",
      subject_name: "",
      summary: "",
      debit_amount: 0,
      credit_amount: 0,
    };
    setVoucherItems((prev) => [...prev, newItem]);
  };

  const updateVoucherItem = (index: number, field: keyof VoucherItemRequest, value: any) => {
    const items = [...voucherItems];
    if (field === "subject_code") {
      const subject = subjects.find((s) => s.code === value);
      items[index] = { ...items[index], subject_code: value, subject_name: subject?.name || "" };
    } else {
      items[index] = { ...items[index], [field]: value };
    }
    setVoucherItems(items);
  };

  const removeVoucherItem = (index: number) => {
    const items = voucherItems.filter((_, i) => i !== index);
    setVoucherItems(items.map((item, i) => ({ ...item, line_number: i + 1 })));
  };

  const validateForm = (): boolean => {
    if (!formData.voucher_number) {
      addNotification({ title: "验证错误", message: "请输入凭证编号", type: "warning" });
      return false;
    }
    if (!formData.voucher_date) {
      addNotification({ title: "验证错误", message: "请输入凭证日期", type: "warning" });
      return false;
    }
    // 更多验证...
    return true;
  };

  const handleSubmit = async () => {
    if (!validateForm()) return;

    setSubmitting(true);
    try {
      const result = await voucherApi.create({
        ...formData,
        items: voucherItems.filter(
          (item) => item.subject_code && (item.debit_amount > 0 || item.credit_amount > 0)
        ),
      });

      if (result.success) {
        addNotification({ title: "成功", message: "凭证创建成功", type: "success" });
        navigate("/finance/vouchers");
      } else {
        addNotification({ title: "失败", message: result.message || "创建失败", type: "error" });
      }
    } catch (err: any) {
      addNotification({ title: "错误", message: err.message || "创建失败", type: "error" });
    } finally {
      setSubmitting(false);
    }
  };

  const handleCancel = () => {
    navigate("/finance/vouchers");
  };

  const formatCurrency = (amount: number) => {
    return `¥${amount.toLocaleString("zh-CN", { minimumFractionDigits: 2 })}`;
  };

  const totalDebit = voucherItems.reduce((sum, item) => sum + (item.debit_amount || 0), 0);
  const totalCredit = voucherItems.reduce((sum, item) => sum + (item.credit_amount || 0), 0);
  const isBalanced = totalDebit === totalCredit && totalDebit > 0;

  // 基本信息Section
  const basicInfoSection = (
    <Card>
      <CardContent>
        <Typography variant="h6" sx={{ fontWeight: 600, mb: 3 }}>
          基本信息
        </Typography>
        <Box sx={{ display: "grid", gridTemplateColumns: { xs: "1fr", md: "repeat(3, 1fr)" }, gap: 3 }}>
          <TextField
            label="凭证编号 *"
            value={formData.voucher_number}
            onChange={(e) => handleFieldChange("voucher_number", e.target.value)}
            placeholder="例如: 记-2024-001"
            required
            fullWidth
            size="small"
          />
          <TextField
            label="凭证日期 *"
            type="date"
            value={formData.voucher_date}
            onChange={(e) => handleFieldChange("voucher_date", e.target.value)}
            required
            fullWidth
            size="small"
            slotProps={{ inputLabel: { shrink: true } }}
          />
          <TextField
            label="会计期间 *"
            value={formData.period}
            onChange={(e) => handleFieldChange("period", e.target.value)}
            placeholder="例如: 2024-01"
            required
            fullWidth
            size="small"
            helperText="格式: YYYY-MM"
          />
        </Box>
      </CardContent>
    </Card>
  );

  // 凭证明细Section
  const voucherItemsSection = (
    <Card>
      <CardContent>
        <Box sx={{ display: "flex", justifyContent: "space-between", alignItems: "center", mb: 3 }}>
          <Typography variant="h6" sx={{ fontWeight: 600 }}>
            凭证明细
          </Typography>
          {!isBalanced && totalDebit > 0 && (
            <Chip size="small" icon={<AlertCircle size={14} />} label="借贷不平衡" color="warning" variant="outlined" />
          )}
          <Button variant="outlined" size="small" startIcon={<Plus size={16} />} onClick={addVoucherItem}>
            添加分录
          </Button>
        </Box>

        <TableContainer component={Paper} variant="outlined">
          <Table>
            <TableHead>
              <TableRow sx={{ bgcolor: "grey.100" }}>
                <TableCell sx={{ fontWeight: 600, width: 80 }}>行号</TableCell>
                <TableCell sx={{ fontWeight: 600 }}>会计科目</TableCell>
                <TableCell sx={{ fontWeight: 600 }}>摘要</TableCell>
                <TableCell sx={{ fontWeight: 600, width: 180, textAlign: "right" }}>借方金额</TableCell>
                <TableCell sx={{ fontWeight: 600, width: 180, textAlign: "right" }}>贷方金额</TableCell>
                <TableCell sx={{ fontWeight: 600, width: 80 }}>操作</TableCell>
              </TableRow>
            </TableHead>
            <TableBody>
              {voucherItems.map((item, index) => (
                <TableRow key={index}>
                  <TableCell><Typography sx={{ fontWeight: 600, color: "text.secondary" }}>{item.line_number}</Typography></TableCell>
                  <TableCell>
                    <Autocomplete
                      size="small"
                      options={subjects}
                      getOptionLabel={(option) => `${option.code} - ${option.name}`}
                      value={subjects.find((s) => s.code === item.subject_code) || null}
                      onChange={(_, newValue) => updateVoucherItem(index, "subject_code", newValue?.code || "")}
                      renderInput={(params) => <TextField {...params} placeholder="选择或搜索科目" size="small" />}
                    />
                  </TableCell>
                  <TableCell>
                    <TextField size="small" placeholder="分录摘要" value={item.summary} onChange={(e) => updateVoucherItem(index, "summary", e.target.value)} />
                  </TableCell>
                  <TableCell>
                    <TextField size="small" type="number" value={item.debit_amount || ""} onChange={(e) => updateVoucherItem(index, "debit_amount", parseFloat(e.target.value) || 0)} sx={{ "& input": { textAlign: "right", color: "error.main" } }} />
                  </TableCell>
                  <TableCell>
                    <TextField size="small" type="number" value={item.credit_amount || ""} onChange={(e) => updateVoucherItem(index, "credit_amount", parseFloat(e.target.value) || 0)} sx={{ "& input": { textAlign: "right", color: "success.main" } }} />
                  </TableCell>
                  <TableCell>
                    <IconButton size="small" color="error" onClick={() => removeVoucherItem(index)} disabled={voucherItems.length <= 1}>
                      <Trash2 size={16} />
                    </IconButton>
                  </TableCell>
                </TableRow>
              ))}
            </TableBody>
          </Table>
        </TableContainer>

        {/* 合计和验证 */}
        <Box sx={{ display: "flex", justifyContent: "flex-end", gap: 4, mt: 3, pt: 2, borderTop: 2, borderColor: "divider" }}>
          <Box sx={{ textAlign: "right" }}>
            <Typography variant="body2" color="text.secondary">借方合计</Typography>
            <Typography variant="h6" sx={{ fontWeight: "bold", color: "error.main" }}>{formatCurrency(totalDebit)}</Typography>
          </Box>
          <Box sx={{ textAlign: "right" }}>
            <Typography variant="body2" color="text.secondary">贷方合计</Typography>
            <Typography variant="h6" sx={{ fontWeight: "bold", color: isBalanced ? "success.main" : "error.main" }}>{formatCurrency(totalCredit)}</Typography>
          </Box>
        </Box>

        {!isBalanced && totalDebit > 0 && (
          <Alert severity="warning" icon={<Calculator size={20} />} sx={{ mt: 2 }}>
            <Typography variant="body2">
              <strong>借贷不平衡:</strong> 借方合计 {formatCurrency(totalDebit)} ≠ 贷方合计 {formatCurrency(totalCredit)}, 差额 {formatCurrency(Math.abs(totalDebit - totalCredit))}
            </Typography>
          </Alert>
        )}

        {isBalanced && (
          <Alert severity="success" icon={<Calculator size={20} />} sx={{ mt: 2 }}>
            <Typography variant="body2"><strong>借贷平衡</strong> - 借方合计 = 贷方合计 = {formatCurrency(totalDebit)}</Typography>
          </Alert>
        )}
      </CardContent>
    </Card>
  );

  // Section配置
  const sections = [
    { id: "basic", title: "基本信息", content: basicInfoSection, collapsible: false },
    { id: "items", title: "凭证明细", content: voucherItemsSection, collapsible: false },
  ];

  return (
    <FormPageTemplate
      title="新建凭证"
      breadcrumbs={[
        { label: "首页", href: "/dashboard" },
        { label: "财务", href: "/finance" },
        { label: "凭证管理", href: "/finance/vouchers" },
        { label: "新建凭证" },
      ]}
      sections={sections}
      onSubmit={handleSubmit}
      onCancel={handleCancel}
      submitText="保存凭证"
      submitDisabled={submitting || !isBalanced}
      submitLoading={submitting}
      submitIcon={<Save size={18} />}
      cancelIcon={<X size={18} />}
      helpText="提示: 借贷必须平衡才能保存凭证"
    />
  );
};

export default VoucherCreatePage;
```

```tsx
import {
  Card,
  CardContent,
  Typography,
  Box,
  TextField,
  Button,
  Chip,
  Table,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
  IconButton,
  Checkbox,
  Stack,
  Pagination,
  FormControl,
  InputLabel,
  Select,
  MenuItem,
} from "@mui/material";
import React, { useState, useEffect } from "react";
import { useNavigate } from "react-router-dom";
import {
  Search,
  Plus,
  Edit,
  Trash2,
  Eye,
  Filter,
} from "lucide-react";
import { ListPageTemplate } from "@/components/templates";
import { crmApi, type Customer } from "@/api/modules/crm/crm";
import { useAppStore } from "@/store";
import type { ColumnDef } from "@tanstack/react-table";
import { EnhancedDataTable } from "@/components/EnhancedDataTable";

const CustomerManagementPage: React.FC = () => {
  const navigate = useNavigate();
  const addNotification = useAppStore((state) => state.addNotification);

  const [customers, setCustomers] = useState<Customer[]>([]);
  const [loading, setLoading] = useState(true);
  const [searchTerm, setSearchTerm] = useState("");
  const [statusFilter, setStatusFilter] = useState<string>("all");
  const [currentPage, setCurrentPage] = useState(1);
  const [pageSize, setPageSize] = useState(10);
  const [total, setTotal] = useState(0);
  const [selectedIds, setSelectedIds] = useState<string[]>([]);

  useEffect(() => {
    loadCustomers();
  }, [currentPage, pageSize, searchTerm, statusFilter]);

  const loadCustomers = async () => {
    try {
      setLoading(true);
      const params: any = { page: currentPage, pageSize };
      if (searchTerm) params.search = searchTerm;
      if (statusFilter !== "all") params.status = statusFilter;

      const response = await crmApi.customers.list(params);
      if (response.success && response.data) {
        setCustomers(response.data.items || []);
        setTotal(response.data.total || 0);
      }
    } catch (error) {
      console.error("加载客户列表失败:", error);
      addNotification({
        type: "error",
        title: "加载失败",
        message: "无法加载客户列表",
      });
    } finally {
      setLoading(false);
    }
  };

  const handleDelete = async (id: string) => {
    if (!window.confirm("确定要删除这个客户吗?")) return;
    try {
      await crmApi.customers.delete(id);
      addNotification({
        type: "success",
        title: "删除成功",
        message: "客户已删除",
      });
      loadCustomers();
    } catch (error) {
      addNotification({
        type: "error",
        title: "删除失败",
        message: "无法删除客户",
      });
    }
  };

  const handleBatchDelete = async () => {
    if (selectedIds.length === 0) return;
    if (!window.confirm(`确定要删除选中的 ${selectedIds.length} 个客户吗?`)) return;
    try {
      await crmApi.customers.batchDelete(selectedIds);
      addNotification({
        type: "success",
        title: "删除成功",
        message: `已删除 ${selectedIds.length} 个客户`,
      });
      setSelectedIds([]);
      loadCustomers();
    } catch (error) {
      addNotification({
        type: "error",
        title: "删除失败",
        message: "批量删除失败",
      });
    }
  };

  const getStatusBadge = (status: string) => {
    const statusMap: Record<string, { label: string; color: "success" | "default" | "warning" | "error" }> = {
      active: { label: "活跃", color: "success" },
      inactive: { label: "非活跃", color: "default" },
      blocked: { label: "已冻结", color: "error" },
    };
    const config = statusMap[status] || { label: status, color: "default" };
    return <Chip size="small" color={config.color} label={config.label} />;
  };

  // 列配置
  const columns: ColumnDef<Customer>[] = [
    {
      accessorKey: "name",
      header: "客户名称",
      cell: ({ row }) => (
        <Typography sx={{ fontWeight: 500, color: "primary.main" }}>
          {row.original.name}
        </Typography>
      ),
    },
    { accessorKey: "code", header: "客户代码", width: 120 },
    { accessorKey: "industry", header: "行业" },
    { accessorKey: "status", header: "状态", cell: ({ row }) => getStatusBadge(row.original.status) },
  ];

  // 行操作
  const rowActions = [
    { label: "查看", icon: <Eye />, onClick: (row: Customer) => navigate(`/crm/customers/${row.id}`) },
    { label: "编辑", icon: <Edit />, onClick: (row: Customer) => navigate(`/crm/customers/${row.id}/edit`) },
    { label: "删除", icon: <Trash2 />, color: "error" as const, onClick: (row: Customer) => handleDelete(row.id) },
  ];

  // 高级筛选
  const advancedFilters = [
    {
      label: "状态",
      key: "status",
      type: "select" as const,
      options: [
        { value: "all", label: "全部状态" },
        { value: "active", label: "活跃" },
        { value: "inactive", label: "非活跃" },
        { value: "blocked", label: "已冻结" },
      ],
      value: statusFilter,
      onChange: setStatusFilter,
    },
  ];

  return (
    <ListPageTemplate
      title="客户列表"
      breadcrumbs={[
        { label: "首页", href: "/dashboard" },
        { label: "客户", href: "/crm" },
        { label: "客户列表" },
      ]}
      columns={columns}
      data={customers}
      loading={loading}
      rowActions={rowActions}
      searchable
      searchPlaceholder="搜索客户名称或代码..."
      searchKey="name"
      enableSelection
      selectedIds={selectedIds}
      onSelectionChange={setSelectedIds}
      selectionKey="id"
      enableAdvancedFilter
      advancedFilters={advancedFilters}
      pagination={{
        page: currentPage,
        pageSize,
        total,
        onPageChange: setCurrentPage,
        onPageSizeChange: setPageSize,
      }}
      actions={[
        {
          label: "新建客户",
          icon: <Plus />,
          onClick: () => navigate("/crm/customers/create"),
        },
      ]}
      onRefresh={loadCustomers}
      exportFileName="customer-report"
      batchActions={
        selectedIds.length > 0
          ? [
              {
                label: "批量删除",
                icon: <Trash2 />,
                color: "error",
                onClick: handleBatchDelete,
              },
            ]
          : []
      }
    />
  );
};

export default CustomerManagementPage;
```

### Q1: 如何处理复杂的数据验证?

```typescript
const validateForm = (): boolean => {
  const errors: string[] = [];

  if (!formData.name) {
    errors.push("名称不能为空");
  }

  if (formData.email && !isValidEmail(formData.email)) {
    errors.push("邮箱格式不正确");
  }

  if (errors.length > 0) {
    addNotification({
      title: "验证错误",
      message: errors.join(", "),
      type: "warning",
    });
    return false;
  }

  return true;
};
```
