## 五、QMS系统详细设计

### 5.1 质量管理标准

#### 5.1.1 数据结构

```typescript
interface QualityStandard {
  id: string;
  name: string;                // 标准名称
  code: string;                // 标准编号
  category: string;            // 标准类别
  version: string;             // 版本号
  description: string;         // 标准描述
  content: string;             // 标准内容
  status: string;              // 状态：启用/禁用/草稿
  effective_date: Date;        // 生效日期
  expiry_date?: Date;          // 失效日期
  created_at: Date;
  updated_at: Date;
}

interface QualityCheck {
  id: string;
  check_no: string;            // 检查单号
  project_id?: string;         // 关联项目
  check_type: string;          // 检查类型
  check_items: CheckItem[];    // 检查项
  inspector: string;           // 检查人
  check_date: Date;            // 检查日期
  location: string;            // 检查地点
  result: string;              // 检查结果
  conclusion: string;          // 结论
  attachments: string[];       // 附件
  status: string;              // 状态
  created_at: Date;
  updated_at: Date;
}

interface CheckItem {
  id: string;
  standard_id: string;         // 关联标准
  item_name: string;           // 检查项名称
  check_method: string;        // 检查方法
  standard_value: string;      // 标准值
  actual_value?: string;       // 实际值
  is_conform: boolean;         // 是否合格
  remarks?: string;            // 备注
}

interface QualityIssue {
  id: string;
  issue_no: string;            // 问题编号
  title: string;               // 问题标题
  description: string;         // 问题描述
  severity: string;            // 严重程度
  category: string;            // 问题类别
  status: string;              // 状态
  assignee: string;            // 负责人
  due_date?: Date;             // 截止日期
  resolution?: string;         // 解决方案
  closed_at?: Date;            // 关闭时间
  created_at: Date;
  updated_at: Date;
}
```

#### 5.1.2 API接口

| 方法 | 路径 | 说明 |
| ------ | ------ | ------ |
| GET | `/api/v1/standards` | 获取标准列表 |
| POST | `/api/v1/standards` | 创建标准 |
| GET | `/api/v1/standards/:id` | 获取单个标准 |
| PUT | `/api/v1/standards/:id` | 更新标准 |
| DELETE | `/api/v1/standards/:id` | 删除标准 |

| GET | `/api/v1/checks/:id` | 获取检查详情 |
| PUT | `/api/v1/checks/:id` | 更新检查 |
| DELETE | `/api/v1/checks/:id` | 删除检查 |

## 六、EHS系统详细设计

### 6.1 环境健康安全

#### 6.1.1 数据结构

```typescript
interface SafetyHazard {
  id: string;
  hazard_no: string;           // 危险源编号
  name: string;                // 危险源名称
  category: string;            // 类别
  level: string;               // 危险等级
  description: string;         // 描述
  location: string;            // 位置
  risk_assessment: RiskAssessment;
  control_measures: string[];  // 控制措施
  status: string;              // 状态
  created_at: Date;
  updated_at: Date;
}

interface RiskAssessment {
  likelihood: number;          // 可能性（1-5）
  severity: number;            // 严重性（1-5）
  risk_value: number;          // 风险值
  risk_level: string;          // 风险等级
  assessment_date: Date;
  assessor: string;
}

interface SafetyCheck {
  id: string;
  check_no: string;            // 检查单号
  check_type: string;          // 检查类型
  check_items: SafetyCheckItem[];
  inspector: string;           // 检查人
  check_date: Date;            // 检查日期
  location: string;            // 检查地点
  result: string;              // 检查结果
  issues: string[];            // 发现问题
  attachments: string[];       // 附件
  status: string;              // 状态
  created_at: Date;
  updated_at: Date;
}

interface SafetyCheckItem {
  id: string;
  hazard_id?: string;          // 关联危险源
  item_name: string;           // 检查项名称
  check_method: string;        // 检查方法
  standard: string;            // 检查标准
  is_conform: boolean;         // 是否符合
  remarks?: string;            // 备注
}

interface EmergencyPlan {
  id: string;
  plan_no: string;             // 预案编号
  name: string;                // 预案名称
  type: string;                // 预案类型
  scope: string;               // 适用范围
  trigger_conditions: string;  // 触发条件
  response_procedures: string; // 响应程序
  responsible_persons: string[];
  contact_info: string;
  drills: DrillRecord[];        // 演练记录
  status: string;              // 状态
  created_at: Date;
  updated_at: Date;
}
```
