## 五、数据结构

### 项目 (Project)
```typescript
interface Project {
  id: string;              // UUID
  name: string;            // 项目名称
  description: string;     // 项目描述
  status: string;          // 状态：立项/进行中/已完成/已终止
  start_date: string;      // 开始时间
  end_date?: string;       // 结束时间
  created_at: string;      // 创建时间
  updated_at: string;       // 更新时间
}
```

### 样品 (Sample)
```typescript
interface Sample {
  id: string;              // UUID
  name: string;            // 样品名称
  description: string;     // 样品描述
  status: string;          // 状态：待测试/测试中/已完成/异常
  project_id?: string;     // 关联项目ID
  received_date: string;   // 接收时间
  created_at: string;      // 创建时间
  updated_at: string;      // 更新时间
}
```
