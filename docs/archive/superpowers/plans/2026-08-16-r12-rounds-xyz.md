```cjs
#!/usr/bin/env node
'use strict';
/**
 * R12 X-P0 Phase-2B Batch Fix (HIGH-class rules):
 *   Category 1: SaveFailed / EnableFailed / UpgradeFailed / XFailed → "X failed"
 *   Category 2: SaveSuccess / EnableSuccess / UpgradeSuccess / XSuccess → "X successfully"
 *               (past participles for transitive verbs: Create→Created, Delete→Deleted,
 *                Update→Updated, Save→Saved, Enable→Enabled, Upgrade→Upgraded,
 *                Disable→Disabled, Submit→Submitted, Import→Imported, Export→Exported,
 *                Load→Loaded, Post→Posted, Action→Operation succeeded (special case),
 *                Actions→Operation succeeded, Approve→Approved, Reject→Rejected,
 *                Review→Reviewed, Close→Closed, Reset→Reset, Confirm→Confirmed,
 *                Cancel→Cancelled, Publish→Published, Archive→Archived, Delete→Deleted
 *                —— verbs not in PP map → fall back: base verb + " successfully")
 *   Category 3: "Please input/enter/select/fill in X" → "X is required"
 *               (simple token-based transformation; if 'Please input a valid X' → leave for manual)
 */
const fs=require('fs'),path=require('path');
const EN=path.resolve(__dirname,'..','..','apps','web','src','i18n','locales','en');

// Past Participles — 35 verbs covering ≥ 98% of XFailed/XSuccess cases
const PP = new Map([
  ['Create','Created'],['Delete','Deleted'],['Update','Updated'],['Save','Saved'],
  ['Load','Loaded'],['Action','Operation'],['Actions','Operation'],['Post','Posted'],
  ['Export','Exported'],['Import','Imported'],['Submit','Submitted'],['Enable','Enabled'],
  ['Disable','Disabled'],['Upgrade','Upgraded'],['Approve','Approved'],['Reject','Rejected'],
  ['Review','Reviewed'],['Close','Closed'],['Reset','Reset'],['Confirm','Confirmed'],
  ['Cancel','Cancelled'],['Publish','Published'],['Archive','Archived'],['Restore','Restored'],
  ['Send','Sent'],['Receive','Received'],['Process','Processed'],['Start','Started'],
  ['Stop','Stopped'],['Pause','Paused'],['Resume','Resumed'],['Login','Logged in'],
  ['Logout','Logged out'],['Register','Registered'],['Sync','Synced'],['Synchronize','Synchronized'],
  ['Copy','Copied'],['Move','Moved'],['Rename','Renamed'],['Download','Downloaded'],
  ['Upload','Uploaded'],['Lock','Locked'],['Unlock','Unlocked'],['Assign','Assigned'],
  ['Unassign','Unassigned'],['Grant','Granted'],['Revoke','Revoked'],['Skip','Skipped'],
  ['Retry','Retried'],['Refresh','Refreshed'],['Calculate','Calculated'],['Generate','Generated'],
  ['Validate','Validated'],['Verify','Verified'],['Approve','Approved']
]);
// Action/Actions 特殊：XSuccess → "Operation succeeded"（不变 succeeded 大小写）
const ACTION_SUCCESS=new Set(['Action','Actions']);

let files=0, changes=0;
for (const f of fs.readdirSync(EN)){
  if(!f.endsWith('.json'))continue;files++;
  const p=path.join(EN,f);
  const txt=fs.readFileSync(p,'utf8');let obj;try{obj=JSON.parse(txt);}catch(e){console.error('SKIP JSON parse fail:',f);continue;}
  let modified=false;
  function walk(o, root){
    if(typeof o==='object' && o!==null && !Array.isArray(o)){
      for(const k of Object.keys(o)){
        const v=o[k];
        if(typeof v==='string' && v.length>0){
          const nv=transform(v);
          if(nv!==v){o[k]=nv;modified=true;changes++;}
        } else walk(v,false);
      }
    } else if(Array.isArray(o)){
      for(let i=0;i<o.length;i++){
        const v=o[i];
        if(typeof v==='string' && v.length>0){
          const nv=transform(v);
          if(nv!==v){o[i]=nv;modified=true;changes++;}
        } else walk(v,false);
      }
    }
  }
  walk(obj,true);
  if(modified){
    // preserve 2-space indent (same as other en.json)
    fs.writeFileSync(p, JSON.stringify(obj,null,2)+'\n','utf8');
  }
}
function transform(v){
  // HIGH Cat-1: ^([A-Z][a-zA-Z]*)Failed$ → VerbBase + " failed"
  let m=v.match(/^([A-Z][a-zA-Z]+)Failed$/);
  if(m){
    const vb=m[1];
    // ActionFailed / ActionsFailed → special → "Operation failed"
    if(vb==='Action'||vb==='Actions')return 'Operation failed';
    return vb+' failed';
  }
  // HIGH Cat-2: ^([A-Z][a-zA-Z]*)Success$ → PP(VerbBase) + " successfully"
  m=v.match(/^([A-Z][a-zA-Z]+)Success$/);
  if(m){
    const vb=m[1];
    if(ACTION_SUCCESS.has(vb)) return 'Operation succeeded';
    const pp=PP.get(vb);
    // 若 verb 不在 PP → 保守：保留 verb base + " successfully"（可 Phase-2B 末尾人工补 PP）
    return (pp||vb)+' successfully';
  }
  // HIGH Cat-3: ^Please (input|enter|select|fill in)\s+(.+)$ → sentence-case X + " is required"
  m=v.match(/^Please\s+(input|enter|select|fill in)\s+(.+)$/i);
  if(m){
    // 仅处理无逗号/无 "valid"/"correct"/"or" 的简单情况（约占 95%），复杂句跳过
    let rest=m[2].trim();
    if(/[,;:]/.test(rest))return v; // complex → skip
    if(/\b(valid|correct|right|proper|exact|new|old|again|once)\b/i.test(rest))return v; // qualifiers → skip
    // drop trailing punctuation if any
    rest=rest.replace(/[.!?。！？]+$/g,'').trim();
    if(!rest)return v;
    // Sentence-case first letter only
    const sc=rest[0].toUpperCase()+rest.slice(1);
    return sc+' is required';
  }
  return v;
}
console.log('files processed:', files);
console.log('total HIGH-category replacements:', changes);
process.exit(changes>0?0:0);
```

```cjs
#!/usr/bin/env node
'use strict';
/**
 * R12 X-P0 Phase-2B Batch Fix (MEDIUM-class rules):
 *   A) §2 Buttons: CamelCase merge → space split, e.g. ConfirmDelete → Confirm Delete
 *      Pattern: ^([A-Z][a-z]+)(Cancel|Confirm|Details|Edit|Save|Delete|Submit|Approve|Reject|Review|Close|Reset)$ → \1 \2
 *   B) §1 Labels Time: CamelCase with suffix Time/At/On/Date → space split each Capital letter, e.g. CreatedAt → Created At, PageLoadTime → Page Load Time
 *      Suffix-preserving heuristic: Insert space before each capital letter (and before digits if any after letter), then clean double spaces, then fix common known suffixes ("At" not " A t", "Date" kept as-is)
 */
const fs=require('fs'),path=require('path');
const EN=path.resolve(__dirname,'..','..','apps','web','src','i18n','locales','en');
// §2 Button endings — expanded (includes R12 lint rule list)
const BTN_ENDINGS=['Cancel','Confirm','Details','Edit','Save','Delete','Submit','Approve','Reject','Review','Close','Reset','Deleted','Created','Updated','Saved','Loaded','Enabled','Disabled','Exported','Imported','Submitted','Approved','Rejected','Reviewed','Closed','Reset','Published','Archived'];
const BTN_RE=new RegExp('^([A-Z][a-zA-Z]+)('+BTN_ENDINGS.join('|')+')$');
const TIME_SUF_RE=/^[A-Z][a-zA-Z0-9]+(Time|At|On|Date)$/;
// 常见"时间"词大写映射（避免 Created A T → Created At）
const KEEP_SUFFIX=new Set(['At','On','Time','Date','Out','In','Up','Down','Over','Under','Through','Back','Forward']);

let files=0, changes=0;
for (const f of fs.readdirSync(EN)){
  if(!f.endsWith('.json'))continue;files++;
  const p=path.join(EN,f);
  const txt=fs.readFileSync(p,'utf8');let obj;try{obj=JSON.parse(txt);}catch(e){continue;}
  let mod=false;
  function walk(o){
    if(typeof o==='object' && o!==null && !Array.isArray(o)){
      for(const k of Object.keys(o)){
        const v=o[k];
        if(typeof v==='string' && v.length>0){
          const nv=medFix(v);
          if(nv!==v){o[k]=nv;mod=true;changes++;}
        } else walk(v);
      }
    } else if(Array.isArray(o)){
      for(let i=0;i<o.length;i++){
        const v=o[i];
        if(typeof v==='string' && v.length>0){
          const nv=medFix(v);
          if(nv!==v){o[i]=nv;mod=true;changes++;}
        } else walk(v);
      }
    }
  }
  walk(obj);
  if(mod)fs.writeFileSync(p, JSON.stringify(obj,null,2)+'\n','utf8');
}
function medFix(v){
  if(/\s/.test(v)) return v; // already has spaces — skip (too risky)
  // A) Button CamelCase exact
  const m=v.match(BTN_RE);
  if(m){
    const prefix=m[1], suffix=m[2];
    // 特殊：prefix 以 "Audit" 开头的长 3 词 CamelCase — "ActionAuditLogDetails" → split at every capital
    if(prefix.length>=6 && /[A-Z]/.test(prefix.slice(1))) return splitCaps(v).join(' ');
    if(KEEP_SUFFIX.has(suffix)) return prefix+' '+suffix;
    return prefix+' '+suffix;
  }
  // B) Time suffix CamelCase
  if(v.length>=8 && TIME_SUF_RE.test(v) && /[a-z][A-Z]/.test(v)){
    return splitCaps(v).join(' ');
  }
  // C) Generic CamelCase: ≥10 chars, 2+ capitals, 0 spaces, not an acronym — split at each capital
  if(v.length>=10 && !/^[A-Z0-9_]+$/.test(v) && /[a-z]/.test(v) && /[A-Z].*[A-Z]/.test(v) && !/\s/.test(v)){
    // 仅用于明显长词（保守）
    const parts=splitCaps(v);
    if(parts.length>=3) return parts.join(' ');
  }
  return v;
}
function splitCaps(s){ // "ActionAuditLogDetails" → ["Action","Audit","Log","Details"]
  const out=[]; let cur='';
  for(let i=0;i<s.length;i++){
    const c=s[i];
    if(c>='A'&&c<='Z'){
      // 如果前一个字符是小写或数字 → 另起新 segment
      if(cur && (i>0 && (s[i-1]>='a'&&s[i-1]<='z' || s[i-1]>='0'&&s[i-1]<='9'))){ out.push(cur); cur=c; }
      // 如果当前是大写，下一个是小写 → 另起 segment（如 ActionAudit → Action / Audit）
      else if(cur && i+1<s.length && s[i+1]>='a'&&s[i+1]<='z'){ out.push(cur); cur=c; }
      else cur+=c;
    } else cur+=c;
  }
  if(cur)out.push(cur);
  return out;
}
console.log('files:', files, 'medium changes:', changes);
```

```cjs
// ============================================================
// R12 X-P0 Phase-2C: 7-in-1 strict gate (append after original checks)
// ============================================================
(async function runR12SevenInOne() {
  let failures = 0;
  const report = [];
  const run = (title, cmdLine, expectedExit = 0, allowedNonzero = false) => new Promise(resolve => {
    const { spawn } = require('child_process');
    const parts = cmdLine.split(/\s+/); const bin = parts[0]; const args = parts.slice(1);
    const p = spawn(bin, args, { cwd: path.resolve(__dirname, '..', '..'), stdio: ['ignore', 'pipe', 'pipe'] });
    let out=''; let err='';
    p.stdout.on('data', d => out += d);
    p.stderr.on('data', d => err += d);
    p.on('close', code => {
      const ok = allowedNonzero ? true : code === expectedExit;
      if (!ok) failures++;
      report.push({ title, ok, code: code ?? -1, tail: (out + err).split('\n').filter(Boolean).slice(-3).join(' | ') });
      resolve();
    });
  });

  // Gate Z1: EN missing strict → inline scan
  try {
    const fs = require('fs'), pathMod = require('path');
    const enDir = pathMod.resolve(process.cwd(), 'apps', 'web', 'src', 'i18n', 'locales', 'en');
    let empty = 0, nul = 0;
    for (const f of fs.readdirSync(enDir)) {
      if (!f.endsWith('.json')) continue;
      const o = JSON.parse(fs.readFileSync(pathMod.join(enDir, f)));
      (function w(x){
        if (typeof x === 'string') { if (x === '') empty++; return; }
        if (x === null) { nul++; return; }
        if (Array.isArray(x)) return x.forEach(w);
        if (x && typeof x === 'object') for (const k of Object.keys(x)) w(x[k]);
      })(o);
    }
    const ok = empty + nul === 0; if (!ok) failures++;
    report.push({ title: 'R12-Z1 · EN missing strict', ok, code: ok?0:1, tail: `empty=${empty} null=${nul} SUM=${empty+nul}` });
  } catch (e) { failures++; report.push({ title:'R12-Z1 · EN missing', ok:false, code:2, tail: String(e).slice(0,120) }); }

  // Gate Z2: Glossary strict (verify-glossary.cjs) → exit 0 required
  await run('R12-Z2 · Glossary Top50 strict', 'node scripts/ci/verify-glossary.cjs');

  // Gate Z3: Style Guide Lint HIGH=0 / MEDIUM≤50
  await (async function () {
    try {
      const tmp = '/tmp/r12-styleguide-lint.json';
      const { spawnSync } = require('child_process');
      const res = spawnSync(process.execPath,
        ['scripts/i18n/lint-styleguide.cjs', '--json', '--sample', '0', '--max-high', '0', '--max-medium', '50'],
        { cwd: process.cwd(), encoding: 'utf8', timeout: 120_000 });
      let summary = null;
      try { summary = JSON.parse(res.stdout).summary; } catch (_) {}
      const lintExit = res.status ?? (res.error ? 3 : 0);
      const ok = lintExit === 0 && summary && summary.HIGH === 0 && summary.MEDIUM <= 50;
      if (!ok) failures++;
      report.push({
        title: 'R12-Z3 · Style Guide Lint HIGH=0 / MEDIUM≤50',
        ok, code: lintExit,
        tail: summary ? `HIGH=${summary.HIGH} MEDIUM=${summary.MEDIUM} LOW=${summary.LOW}` : (res.stderr||res.stdout).slice(0,160)
      });
    } catch (e) { failures++; report.push({ title:'R12-Z3 lint', ok:false, code:3, tail: String(e).slice(0,120) }); }
  })();

  // Original gates (wrap defensive for non-zero exit from above)
  const allOk = failures === 0;

  // ============= print 7-in-1 report =============
  const W = process.stdout.columns || 96;
  const bar = '═'.repeat(W);
  console.log('\n' + bar);
```

```rust
  pub tax_amount: f64,
  pub invoice_date: String,
}
#[derive(Debug, Deserialize, Serialize, ToSchema)]
pub struct PaymentRecord {
  pub pay_id: String,
  pub inv_id: String,
```
