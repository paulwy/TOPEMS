# TOPEMS 页面实施快速参考卡片

## 四种页面模版速查

### 1. DashboardTemplate (仪表盘)

```tsx
import { DashboardTemplate } from "@/components/templates";

<DashboardTemplate
  title="财务概览"
  breadcrumbs={[
    { label: '首页', href: '/dashboard' },
    { label: '财务' }
  ]}
  statsCards={<Cards />}
  quickActions={<Actions />}
  recentItems={<Recent />}
  chartsArea={<Charts />}
/>
```

**适用**: 模块首页、数据统计
**路由**: `/module`

---

### 2. DetailPageTemplate (详情页)

```tsx
import { DetailPageTemplate } from "@/components/templates";

<DetailPageTemplate
  title="凭证编号"
  subtitle="期间: 2024-01"
  breadcrumbs={[...]}
  backHref="/finance/vouchers"
  tabs={[
    { id: "basic", label: "基本信息", content: <BasicInfo /> },
    { id: "items", label: "凭证明细", badge: 3, content: <Items /> },
  ]}
  infoCard={{
    avatar: <FileText />,
    title: "凭证号",
    subtitle: "期间",
    tags: [<Chip />],
    stats: [{ label, value }],
  }}
  onEdit={handleEdit}
  onDelete={handleDelete}
/>
```

**适用**: 查看单条数据详情
**路由**: `/module/entities/:id`

---

### 3. FormPageTemplate (表单页)

```tsx
import { FormPageTemplate } from "@/components/templates";

<FormPageTemplate
  title="新建凭证"
  breadcrumbs={[...]}
  sections={[
    { id: "basic", title: "基本信息", content: <BasicForm /> },
    { id: "items", title: "凭证明细", content: <ItemsForm /> },
  ]}
  onSubmit={handleSubmit}
  onCancel={handleCancel}
  submitText="保存凭证"
  submitDisabled={false}
  submitLoading={false}
/>
```

**适用**: 新建/编辑数据
**路由**: `/module/entities/create` 或 `/module/entities/:id/edit`

---

### 4. ListPageTemplate (列表页)

```tsx
import { ListPageTemplate } from "@/components/templates";

<ListPageTemplate
  title="客户列表"
  breadcrumbs={[...]}
  columns={[
    { accessorKey: "name", header: "客户名称" },
    { accessorKey: "status", header: "状态" },
  ]}
  data={customers}
  loading={false}
  rowActions={[
    { label: "查看", icon: <Eye />, onClick: (row) => navigate(`/crm/customers/${row.id}`) },
    { label: "编辑", icon: <Edit />, onClick: (row) => navigate(`/crm/customers/${row.id}/edit`) },
    { label: "删除", icon: <Trash2 />, color: "error", onClick: handleDelete },
  ]}
  searchable
  searchPlaceholder="搜索..."
  enableSelection
  selectedIds={selectedIds}
  onSelectionChange={setSelectedIds}
  pagination={{ page, pageSize, total, onPageChange }}
  actions={[{ label: "新建", icon: <Plus />, onClick: () => navigate("/create") }]}
/>
```

**适用**: 数据列表展示
**路由**: `/module/entities`

## 常用UI组件速查

### 状态徽章

```tsx
const getStatusBadge = (status: string) => {
  const config = {
    draft: { label: "草稿", color: "default" },
    active: { label: "活跃", color: "success" },
    pending: { label: "待审核", color: "warning" },
    approved: { label: "已审核", color: "info" },
    error: { label: "错误", color: "error" },
  }[status] || { label: status, color: "default" };

  return <Chip size="small" color={config.color} label={config.label} />;
};
```

### 格式化函数

```tsx
// 货币格式化
const formatCurrency = (amount: number) =>
  `¥${amount.toLocaleString("zh-CN", { minimumFractionDigits: 2 })}`;

// 日期格式化
const formatDate = (dateStr?: string) =>
  dateStr ? new Date(dateStr).toLocaleDateString("zh-CN") : "-";

// 数字格式化
const formatNumber = (num: number) =>
  num.toLocaleString("zh-CN");
```

### 确认对话框

```tsx
const handleDelete = async (id: string) => {
  if (!confirm("确定要删除吗?此操作不可撤销。")) return;

  try {
    await api.delete(id);
    addNotification({
      title: "成功",
      message: "删除成功",
      type: "success",
    });
  } catch (err) {
    addNotification({
      title: "错误",
      message: "删除失败",
      type: "error",
    });
  }
};
```
